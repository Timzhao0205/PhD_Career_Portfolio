dois = [r["doi"].lower() for r in ROWS]
assert len(dois) == len(set(dois)), "duplicate doi"
titles = [r["title"].strip().lower() for r in ROWS]
assert len(titles) == len(set(titles)), "duplicate title"

HEADER = ["source_id","citation","title","authors","year","venue","doi","url",
          "source_type","peer_review_status","quality_tier","topic_tags",
          "claims_supported","verification_basis","access_level","notes"]

def make_citation(r):
    authors = r["authors"]
    title = r["title"]
    venue = r["venue"]
    year = r["year"]
    vol = r.get("vol", "")
    issue = r.get("issue", "")
    pages = r.get("pages", "")
    parts = [venue]
    volissue = ""
    if vol:
        volissue = f"vol. {vol}"
        if issue:
            volissue += f", no. {issue}"
    if volissue:
        parts.append(volissue)
    if pages:
        parts.append(f"pp. {pages}")
    tail = ", ".join(parts)
    return f'{authors}, "{title}," {tail}, {year}.'

with open("evidence/10A_GAN_WBG_SOURCES.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.DictWriter(f, fieldnames=HEADER)
    w.writeheader()
    for r in ROWS:
        doi_lc = r["doi"].lower()
        vb = basis_lead(r["batch"], doi_lc) if r["id"] in ABSTRACT_VERIFIED_IDS else basis(r["batch"], doi_lc)
        w.writerow({
            "source_id": r["id"],
            "citation": make_citation(r),
            "title": r["title"],
            "authors": r["authors"],
            "year": r["year"],
            "venue": r["venue"],
            "doi": doi_lc,
            "url": f"https://doi.org/{doi_lc}",
            "source_type": r["source_type"],
            "peer_review_status": "verified_peer_reviewed",
            "quality_tier": r["quality_tier"],
            "topic_tags": r["tags"],
            "claims_supported": r["claims"],
            "verification_basis": vb,
            "access_level": r["access"],
            "notes": r["note"],
        })

print(f"Wrote {len(ROWS)} rows to evidence/10A_GAN_WBG_SOURCES.csv")
