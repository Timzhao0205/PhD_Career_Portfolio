    assert len(ROWS) >= 10, "stage gate: at least 10 technologies"
    for r in ROWS:
        missing = [c for c in HEADER if c not in r]
        extra = [c for c in r if c not in HEADER]
        assert not missing, f"{r.get('technology')}: missing {missing}"
        assert not extra, f"{r.get('technology')}: extra {extra}"
    techs = [r["technology"] for r in ROWS]
    assert len(techs) == len(set(techs)), "duplicate technology rows"
    out = os.path.normpath(OUT)
    with open(out, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=HEADER, quoting=csv.QUOTE_ALL)
        w.writeheader()
        for r in ROWS:
            w.writerow(r)
    print(f"wrote {out}: {len(ROWS)} rows x {len(HEADER)} cols")


if __name__ == "__main__":
    main()
