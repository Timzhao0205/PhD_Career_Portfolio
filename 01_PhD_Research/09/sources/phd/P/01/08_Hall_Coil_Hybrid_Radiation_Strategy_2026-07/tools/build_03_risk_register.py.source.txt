    for row in ROWS:

    assert len(ids) == len(set(ids)), "duplicate risk_id"
    with open(out_path, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f, quoting=csv.QUOTE_ALL)
        w.writerow(HEADER)
        w.writerows(ROWS)
    print(f"wrote {out_path}: {len(ROWS)} rows")


if __name__ == "__main__":
